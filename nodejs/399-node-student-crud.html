399. Create a Student Management CRUD application

A CRUD API specifically modeling student records, with fields relevant to academic data.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const app = express();
app.use(express.json());
let students = [];
let nextId = 1;

app.post('/students', (req, res) => {
  const student = { id: nextId++, name: req.body.name, course: req.body.course, cgpa: req.body.cgpa };
  students.push(student);
  res.status(201).json(student);
});
app.get('/students', (req, res) => res.json(students));
app.put('/students/:id', (req, res) => {
  const student = students.find(s => s.id === Number(req.params.id));
  if (!student) return res.status(404).json({ error: 'Student not found' });
  Object.assign(student, req.body);
  res.json(student);
});
app.delete('/students/:id', (req, res) => {
  students = students.filter(s => s.id !== Number(req.params.id));
  res.status(204).send();
});

app.listen(3000, () => console.log('Student CRUD API running on port 3000'));
