377. Implement route parameters and query parameters

Demonstrates capturing dynamic values from the URL path (route params) and from the query string (?key=value).

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const app = express();

// Route parameter: /students/5  ->  req.params.id === '5'
app.get('/students/:id', (req, res) => {
  res.send(`Fetching student with ID: ${req.params.id}`);
});

// Query parameters: /search?course=CSE&year=2  ->  req.query.course, req.query.year
app.get('/search', (req, res) => {
  const { course, year } = req.query;
  res.send(`Searching for course=${course}, year=${year}`);
});

app.listen(3000, () => console.log('Server running on port 3000'));
