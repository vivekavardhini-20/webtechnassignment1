369. Demonstrate Callbacks, Promises, and Async/Await in Node.js

Demonstrates the same asynchronous file-read operation written three ways: callback style, Promise style, and async/await style.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const fs = require('fs');
const fsPromises = require('fs').promises;

// 1) Callback style
fs.readFile('data.txt', 'utf8', (err, data) => {
  if (err) return console.error('Callback error:', err.message);
  console.log('Callback result:', data);
});

// 2) Promise style
fsPromises.readFile('data.txt', 'utf8')
  .then(data => console.log('Promise result:', data))
  .catch(err => console.error('Promise error:', err.message));

// 3) Async/Await style
async function readFileAsync() {
  try {
    const data = await fsPromises.readFile('data.txt', 'utf8');
    console.log('Async/await result:', data);
  } catch (err) {
    console.error('Async/await error:', err.message);
  }
}
readFileAsync();
