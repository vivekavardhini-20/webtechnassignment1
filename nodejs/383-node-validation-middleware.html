383. Implement request validation middleware

Demonstrates a middleware that checks incoming request data before allowing it to reach the route handler.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const app = express();
app.use(express.json());

function validateStudent(req, res, next) {
  const { name, cgpa } = req.body;
  if (!name || typeof cgpa !== 'number') {
    return res.status(400).json({ error: 'name (string) and cgpa (number) are required' });
  }
  next(); // validation passed
}

app.post('/students', validateStudent, (req, res) => {
  res.status(201).json({ message: 'Student created', student: req.body });
});

app.listen(3000, () => console.log('Server running on port 3000'));
