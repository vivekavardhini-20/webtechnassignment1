413. Implement Authorization Middleware

A generic, reusable authorization middleware factory that can check for any of several allowed roles.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
function authorize(...allowedRoles) {
  return (req, res, next) => {
    if (!allowedRoles.includes(req.user.role)) {
      return res.status(403).json({ error: 'Access denied - insufficient permissions' });
    }
    next();
  };
}

// Usage: allow both 'admin' AND 'moderator' roles
app.delete('/posts/:id', verifyToken, authorize('admin', 'moderator'), (req, res) => {
  res.json({ message: 'Post deleted' });
});
