378. Create multiple routes using Express Router

Demonstrates grouping related routes together using express.Router() as a mini, mountable app.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
// ---- routes/products.js ----
const express = require('express');
const router = express.Router();

router.get('/', (req, res) => res.send('List all products'));
router.get('/:id', (req, res) => res.send(`Product details for ID ${req.params.id}`));
router.post('/', (req, res) => res.send('Create a new product'));

module.exports = router;

// ---- server.js ----
const express = require('express');
const app = express();
const productRoutes = require('./routes/products');

app.use('/products', productRoutes); // all routes above are now prefixed with /products
app.listen(3000, () => console.log('Server running on port 3000'));
