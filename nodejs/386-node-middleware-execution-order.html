386. Demonstrate the execution order of middleware

Demonstrates that Express middleware runs in the exact order it is registered, via a chain of next() calls.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const app = express();

app.use((req, res, next) => { console.log('Middleware 1'); next(); });
app.use((req, res, next) => { console.log('Middleware 2'); next(); });
app.get('/', (req, res, next) => { console.log('Route handler'); next(); }, (req, res) => {
  console.log('Final middleware after route handler');
  res.send('Check your server console to see the execution order!');
});

app.listen(3000, () => console.log('Server running on port 3000'));
// Console output when visiting "/": Middleware 1 -> Middleware 2 -> Route handler -> Final middleware
