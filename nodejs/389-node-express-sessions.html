389. Implement session management in Express

Demonstrates setting up server-side sessions using the express-session package, which tracks logged-in users across requests.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const session = require('express-session'); // npm install express-session
const app = express();

app.use(session({
  secret: 'my-secret-key',
  resave: false,
  saveUninitialized: true,
  cookie: { maxAge: 60000 * 30 } // 30 minutes
}));

app.get('/visit', (req, res) => {
  req.session.views = (req.session.views || 0) + 1;
  res.send(`You have visited this page ${req.session.views} time(s) in this session.`);
});

app.listen(3000, () => console.log('Server running on port 3000'));
