414. Create a basic RESTful API using Express.js

A minimal REST API skeleton following REST conventions: plural resource nouns and HTTP verbs mapping to actions.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const app = express();
app.use(express.json());

app.get('/api/books', (req, res) => res.json({ message: 'GET all books' }));
app.get('/api/books/:id', (req, res) => res.json({ message: `GET book ${req.params.id}` }));
app.post('/api/books', (req, res) => res.status(201).json({ message: 'POST create book' }));
app.put('/api/books/:id', (req, res) => res.json({ message: `PUT update book ${req.params.id}` }));
app.delete('/api/books/:id', (req, res) => res.status(204).send());

app.listen(3000, () => console.log('REST API running on port 3000'));
