396. Update database records using Node.js

Demonstrates updating an existing document using Mongoose's findByIdAndUpdate().

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const Student = require('./models/Student');

async function updateStudent(id) {
  const updated = await Student.findByIdAndUpdate(
    id,
    { cgpa: 9.0 },
    { new: true } // return the updated document, not the old one
  );
  console.log('Updated student:', updated);
}
updateStudent('someId');
