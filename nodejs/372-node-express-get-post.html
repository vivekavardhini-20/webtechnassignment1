372. Handle GET and POST requests using Express

Demonstrates defining both a GET route (to fetch data) and a POST route (to receive data) in Express.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const app = express();
app.use(express.json()); // needed to parse JSON in POST request bodies

app.get('/api/students', (req, res) => {
  res.json([{ id: 1, name: 'Shaik Abdulkalam' }, { id: 2, name: 'Priya Sharma' }]);
});

app.post('/api/students', (req, res) => {
  const newStudent = req.body; // e.g. { name: "Rahul" }
  console.log('Received new student:', newStudent);
  res.status(201).json({ message: 'Student added', student: newStudent });
});

app.listen(3000, () => console.log('Server running on port 3000'));
