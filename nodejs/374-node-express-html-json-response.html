374. Send HTML and JSON responses using Express

Demonstrates the difference between res.send() for HTML/text and res.json() for structured JSON data.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const app = express();

app.get('/html-page', (req, res) => {
  res.send('<h1>This is an HTML response</h1><p>Sent using res.send()</p>');
});

app.get('/json-data', (req, res) => {
  res.json({ name: 'Shaik Abdulkalam', course: 'CSE AI/ML', cgpa: 8.28 });
});

app.listen(3000, () => console.log('Server running on port 3000'));
