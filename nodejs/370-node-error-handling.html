370. Implement Error Handling in a Node.js application

Demonstrates handling synchronous errors with try/catch, asynchronous errors with error-first callbacks, and unhandled promise rejections.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
// Synchronous error handling
try {
  const result = JSON.parse('{ invalid json ');
} catch (err) {
  console.error('Caught sync error:', err.message);
}

// Error-first callback convention (Node.js standard pattern)
function riskyAsyncTask(callback) {
  setTimeout(() => callback(new Error('Something went wrong')), 500);
}
riskyAsyncTask((err, result) => {
  if (err) { console.error('Callback error:', err.message); return; }
  console.log('Result:', result);
});

// Catching unhandled promise rejections globally
process.on('unhandledRejection', (reason) => {
  console.error('Unhandled Rejection:', reason);
});
