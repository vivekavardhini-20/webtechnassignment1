381. Create and implement custom middleware

Demonstrates writing a custom Express middleware function and applying it with app.use().

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const app = express();

// A custom middleware function: (req, res, next)
function requestTimeLogger(req, res, next) {
  req.requestTime = new Date().toISOString();
  console.log(`Request received at ${req.requestTime}`);
  next(); // MUST call next() to pass control to the next middleware/route
}

app.use(requestTimeLogger); // applies to every route below

app.get('/', (req, res) => res.send(`Handled at ${req.requestTime}`));
app.listen(3000, () => console.log('Server running on port 3000'));
