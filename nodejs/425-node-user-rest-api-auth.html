425. Create a User REST API with Authentication and Authorization

A complete REST API combining JWT authentication and role-based authorization to fully protect user-management endpoints.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const jwt = require('jsonwebtoken');
const app = express();
app.use(express.json());
const SECRET_KEY = 'my-jwt-secret';

function verifyToken(req, res, next) {
  const token = req.headers.authorization?.split(' ')[1];
  if (!token) return res.status(401).json({ error: 'No token' });
  jwt.verify(token, SECRET_KEY, (err, decoded) => {
    if (err) return res.status(403).json({ error: 'Invalid token' });
    req.user = decoded; next();
  });
}
function requireAdmin(req, res, next) {
  if (req.user.role !== 'admin') return res.status(403).json({ error: 'Admins only' });
  next();
}

app.get('/api/users', verifyToken, requireAdmin, async (req, res) => res.json(await User.find()));
app.get('/api/users/me', verifyToken, (req, res) => res.json({ username: req.user.username }));

app.listen(3000, () => console.log('User REST API (auth + authz) running on port 3000'));
