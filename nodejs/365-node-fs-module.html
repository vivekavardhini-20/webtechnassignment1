365. Demonstrate the File System (fs) module

Demonstrates reading from and writing to a file using Node's built-in 'fs' module, both synchronously and asynchronously.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const fs = require('fs');

// Asynchronous write (non-blocking)
fs.writeFile('notes.txt', 'Hello from Node.js fs module!', (err) => {
  if (err) throw err;
  console.log('File written successfully (async).');

  // Asynchronous read
  fs.readFile('notes.txt', 'utf8', (err, data) => {
    if (err) throw err;
    console.log('File content (async):', data);
  });
});

// Synchronous version (blocking - simpler but pauses execution)
fs.writeFileSync('notes-sync.txt', 'Written synchronously');
const syncData = fs.readFileSync('notes-sync.txt', 'utf8');
console.log('File content (sync):', syncData);
