373. Serve static files using Express

Demonstrates serving an entire folder (e.g. images, CSS, client-side JS) as static assets using express.static().

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const path = require('path');
const app = express();

// Any file inside the 'public' folder becomes directly accessible,
// e.g. public/style.css is served at http://localhost:3000/style.css
app.use(express.static(path.join(__dirname, 'public')));

app.get('/', (req, res) => {
  res.sendFile(path.join(__dirname, 'public', 'index.html'));
});

app.listen(3000, () => console.log('Static file server running on port 3000'));
