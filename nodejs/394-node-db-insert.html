394. Insert data into a database using Node.js

Demonstrates creating and saving a new document to the database using a Mongoose model.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const Student = require('./models/Student');

async function addStudent() {
  const newStudent = new Student({ name: 'Shaik Abdulkalam', course: 'CSE AI/ML', cgpa: 8.28 });
  const saved = await newStudent.save();
  console.log('Inserted student:', saved);
}
addStudent();
