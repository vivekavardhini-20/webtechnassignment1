366. Demonstrate the Path (path) module

Demonstrates common methods of the 'path' module for working with file and directory paths across operating systems.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const path = require('path');

const filePath = '/users/shaik/projects/app/index.js';

console.log('basename:', path.basename(filePath));       // index.js
console.log('dirname:', path.dirname(filePath));          // /users/shaik/projects/app
console.log('extname:', path.extname(filePath));          // .js
console.log('join:', path.join('folder', 'subfolder', 'file.txt'));
console.log('resolve:', path.resolve('folder', 'file.txt'));
console.log('parse:', JSON.stringify(path.parse(filePath), null, 2));
