408. Implement JWT-based Authentication

Demonstrates issuing a JSON Web Token on login and verifying it on subsequent requests.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const jwt = require('jsonwebtoken'); // npm install jsonwebtoken
const SECRET_KEY = 'my-jwt-secret';

app.post('/login', (req, res) => {
  const { username, password } = req.body;
  // (assume credentials already verified above)
  const token = jwt.sign({ username }, SECRET_KEY, { expiresIn: '1h' });
  res.json({ token });
});

function verifyToken(req, res, next) {
  const authHeader = req.headers['authorization']; // "Bearer <token>"
  const token = authHeader && authHeader.split(' ')[1];
  if (!token) return res.status(401).json({ error: 'No token provided' });

  jwt.verify(token, SECRET_KEY, (err, decoded) => {
    if (err) return res.status(403).json({ error: 'Invalid token' });
    req.user = decoded;
    next();
  });
}
