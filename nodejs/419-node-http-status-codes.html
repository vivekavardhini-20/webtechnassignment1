419. Implement proper HTTP Status Codes in REST APIs

Demonstrates using the correct HTTP status code for each situation, instead of always returning 200.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
app.get('/api/students/:id', (req, res) => {
  const student = students.find(s => s.id === Number(req.params.id));
  if (!student) return res.status(404).json({ error: 'Not found' });  // 404 Not Found
  res.status(200).json(student);                                      // 200 OK
});
app.post('/api/students', (req, res) => {
  if (!req.body.name) return res.status(400).json({ error: 'Bad request' }); // 400 Bad Request
  res.status(201).json({ message: 'Created' });                              // 201 Created
});
app.get('/api/admin-only', (req, res) => res.status(403).json({ error: 'Forbidden' })); // 403 Forbidden
// Other common codes: 401 Unauthorized, 500 Internal Server Error
