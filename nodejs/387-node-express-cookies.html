387. Create and manage cookies using Express

Demonstrates setting a cookie on the client using the cookie-parser package.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const cookieParser = require('cookie-parser'); // npm install cookie-parser
const app = express();
app.use(cookieParser());

app.get('/set-cookie', (req, res) => {
  res.cookie('username', 'shaik_abdulkalam', { maxAge: 3600000 });
  res.send('Cookie has been set!');
});

app.get('/read-cookie', (req, res) => {
  res.send('Cookie value: ' + req.cookies.username);
});

app.listen(3000, () => console.log('Server running on port 3000'));
