403. Implement input validation for CRUD operations

Demonstrates validating request data before performing a CRUD create/update operation, rejecting bad input early.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
function validateStudentInput(req, res, next) {
  const { name, cgpa } = req.body;
  const errors = [];
  if (!name || name.trim() === '') errors.push('Name is required');
  if (cgpa === undefined || cgpa < 0 || cgpa > 10) errors.push('CGPA must be between 0 and 10');
  if (errors.length > 0) return res.status(400).json({ errors });
  next();
}

app.post('/students', validateStudentInput, (req, res) => {
  // req.body is guaranteed valid here
  res.status(201).json({ message: 'Student created', student: req.body });
});
