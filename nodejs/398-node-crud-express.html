398. Implement CRUD operations using Node.js and Express

Demonstrates all four CRUD operations (Create, Read, Update, Delete) as Express routes over an in-memory array.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const app = express();
app.use(express.json());

let students = [{ id: 1, name: 'Shaik Abdulkalam' }];

app.get('/students', (req, res) => res.json(students));                          // Read all
app.get('/students/:id', (req, res) => {                                          // Read one
  const s = students.find(s => s.id === Number(req.params.id));
  s ? res.json(s) : res.status(404).send('Not found');
});
app.post('/students', (req, res) => {                                             // Create
  const newStudent = { id: students.length + 1, ...req.body };
  students.push(newStudent);
  res.status(201).json(newStudent);
});
app.put('/students/:id', (req, res) => {                                          // Update
  const s = students.find(s => s.id === Number(req.params.id));
  if (!s) return res.status(404).send('Not found');
  Object.assign(s, req.body);
  res.json(s);
});
app.delete('/students/:id', (req, res) => {                                       // Delete
  students = students.filter(s => s.id !== Number(req.params.id));
  res.status(204).send();
});

app.listen(3000, () => console.log('CRUD API running on port 3000'));
