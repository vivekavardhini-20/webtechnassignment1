417. Implement a PUT/PATCH REST API

Demonstrates the difference between PUT (replaces the whole resource) and PATCH (updates only given fields).

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
// PUT - expects the full resource representation
app.put('/api/students/:id', (req, res) => {
  const student = students.find(s => s.id === Number(req.params.id));
  if (!student) return res.status(404).json({ error: 'Not found' });
  student.name = req.body.name; student.course = req.body.course; student.cgpa = req.body.cgpa;
  res.json(student);
});

// PATCH - updates only the fields provided
app.patch('/api/students/:id', (req, res) => {
  const student = students.find(s => s.id === Number(req.params.id));
  if (!student) return res.status(404).json({ error: 'Not found' });
  Object.assign(student, req.body); // merge only provided fields
  res.json(student);
});
