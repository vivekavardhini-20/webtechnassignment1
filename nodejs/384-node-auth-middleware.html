384. Implement authentication middleware

Demonstrates a middleware that checks for a valid auth token before allowing access to protected routes.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const app = express();

function requireAuth(req, res, next) {
  const token = req.headers['authorization'];
  if (!token || token !== 'Bearer secret-token-123') {
    return res.status(401).json({ error: 'Unauthorized - valid token required' });
  }
  next();
}

app.get('/public', (req, res) => res.send('Anyone can see this'));
app.get('/private', requireAuth, (req, res) => res.send('Only authenticated users see this'));

app.listen(3000, () => console.log('Server running on port 3000'));
