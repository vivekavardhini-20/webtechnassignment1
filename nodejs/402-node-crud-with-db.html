402. Implement CRUD operations with database connectivity

Demonstrates the same CRUD pattern as before, but backed by a real MongoDB database via Mongoose instead of an in-memory array.

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
const express = require('express');
const Student = require('./models/Student'); // Mongoose model
const app = express();
app.use(express.json());

app.post('/students', async (req, res) => {
  const student = await Student.create(req.body);
  res.status(201).json(student);
});
app.get('/students', async (req, res) => {
  const students = await Student.find();
  res.json(students);
});
app.put('/students/:id', async (req, res) => {
  const updated = await Student.findByIdAndUpdate(req.params.id, req.body, { new: true });
  res.json(updated);
});
app.delete('/students/:id', async (req, res) => {
  await Student.findByIdAndDelete(req.params.id);
  res.status(204).send();
});

app.listen(3000, () => console.log('DB-backed CRUD API running on port 3000'));
