367. Create and use custom Node.js modules

Demonstrates exporting functions from one file (a custom module) and importing them into another using module.exports and require().

⚠ This is server-side Node.js code. It cannot run inside a static browser HTML file — copy it into a .js file and run it with node filename.js (after npm installing any packages used).
// ---- File: mathUtils.js (the custom module) ----
function add(a, b) { return a + b; }
function multiply(a, b) { return a * b; }
module.exports = { add, multiply };

// ---- File: app.js (uses the custom module) ----
const mathUtils = require('./mathUtils');
console.log('add(5, 3) =', mathUtils.add(5, 3));
console.log('multiply(5, 3) =', mathUtils.multiply(5, 3));
